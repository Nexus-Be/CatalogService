1. Vi tri file va cau hinh:
   - Config file: Catalog.Api/appsettings.json
     Cau hinh:
     "RabbitMq": {
       "UseInMemory": false,
       "Host": "localhost",
       "Port": 5672,
       "User": "guest",
       "Password": "guest"
     }
   - Event Contract: Catalog.Application/Common/Contracts/Events/CategoryCreatedEvent.cs
     public record CategoryCreatedEvent
     {
         public Guid CategoryId { get; init; }
         public string Name { get; init; } = string.Empty;
         public string? Description { get; init; }
         public Guid? ParentId { get; init; }
         public short Level { get; init; }
         public int SortOrder { get; init; }
         public DateTime CreatedAt { get; init; }
     }
   - Consumer: Catalog.Infrastructure/Consumers/CategoryCreatedConsumer.cs
     public sealed class CategoryCreatedConsumer : IConsumer<CategoryCreatedEvent>
     {
         public Task Consume(ConsumeContext<CategoryCreatedEvent> context)
         {
             var msg = context.Message;
             _logger.LogInformation("[MassTransit] CategoryCreatedEvent received: ID={CategoryId}, Name={Name}", msg.CategoryId, msg.Name);
             return Task.CompletedTask;
         }
     }
   - Dang ky Dependency Injection: Catalog.Infrastructure/DependencyInjection.cs
     Dong: 23-48
     Code setup:
     services.AddMassTransit(x =>
     {
         x.SetKebabCaseEndpointNameFormatter();
         x.AddConsumer<CategoryCreatedConsumer>();
         x.UsingRabbitMq((ctx, cfg) =>
         {
             cfg.Host(rabbitHost, (ushort)rabbitPort, "/", h =>
             {
                 h.Username(rabbitUser);
                 h.Password(rabbitPass);
             });
             cfg.ConfigureEndpoints(ctx);
         });
     });

2. Co che phat va nhan su kien (Publish / Consume Flow):
   - Khi tao moi Category thanh cong qua API POST /api/v1/categories (CAT-15):
     Inject: IPublishEndpoint _publishEndpoint
     Goi lenh:
     await _publishEndpoint.Publish(new CategoryCreatedEvent
     {
         CategoryId = category.Id,
         Name = category.Name,
         Description = category.Description,
         ParentId = category.ParentId,
         Level = category.Level,
         SortOrder = category.SortOrder,
         CreatedAt = category.CreatedAt.UtcDateTime
     }, cancellationToken);
   - MassTransit tu dong serialize object thanh JSON, dinh tuyen qua RabbitMQ Exchange "category-created"
   - CategoryCreatedConsumer nhan duoc message, ghi nhan log xac nhan su kien da phat va nhan thanh cong.
   - Luu y quan trong ve Outbox: Da go bo co che EF Core Outbox Polling de loai bo vong lap vo tan retry ket noi database khi test RabbitMQ local.
